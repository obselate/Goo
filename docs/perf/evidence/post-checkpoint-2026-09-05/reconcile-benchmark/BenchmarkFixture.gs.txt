package Goo

import System
import System.Collections.Generic
import System.Diagnostics

public class ReconcileBenchmarkFixture {
  shared {
    private var sink int64

    public func Run() {
      let warmup = environmentCount("GOO_RECON_WARMUP", 500, 100000)
      let samples = environmentCount("GOO_RECON_SAMPLES", 3000, 100000)
      let styleBatch = environmentCount("GOO_RECON_STYLE_BATCH", 4096, 1048576)
      let orderSeed = environmentCount("GOO_RECON_ORDER_SEED", 0, 1000000)
      if samples <= 0 || styleBatch <= 0 {
        throw InvalidOperationException("benchmark samples and style batch must be positive")
      }
      verifyStyleSemantics()
      let caseCount = 13
      let start = orderSeed % caseCount
      var index int32
      while index < caseCount {
        runCase((start + index) % caseCount, warmup, samples, styleBatch)
        index++
      }
      Console.WriteLine("reconcile-benchmark-complete: cases=" + caseCount.ToString()
        + " sink=" + sink.ToString())
    }

    private func runCase(index int32, warmup int32, samples int32, styleBatch int32) {
      switch index {
        case 0 { runStyle("nil-equal", 0, warmup, samples, styleBatch) }
        case 1 { runStyle("numeric-mismatch", 1, warmup, samples, styleBatch) }
        case 2 { runStyle("shared-text", 2, warmup, samples, styleBatch) }
        case 3 { runStyle("distinct-equal-text", 3, warmup, samples, styleBatch) }
        default {
          let keyed = index - 4
          let count = switch keyed / 3 {
            case 0: 10
            case 1: 100
            case 2: 1000
            default: throw InvalidOperationException("invalid benchmark case")
          }
          let name = switch keyed % 3 {
            case 0: "stable"
            case 1: "reorder"
            case 2: "remove-add"
            default: throw InvalidOperationException("invalid keyed case")
          }
          runKeyed(name, count, warmup, samples)
        }
      }
    }

    private func runStyle(name string, kind int32, warmup int32, samples int32, batch int32) {
      let left = [64]StyleEntry
      let right = [64]StyleEntry
      var index int32
      while index < left.Length {
        let numeric = float32(index + 1)
        var leftPayload object?
        var rightPayload object?
        var rightD = numeric + 3.0F
        if kind == 1 {
          rightD = rightD + 1.0F
        } else if kind == 2 {
          let shared = String.Concat("shared-", index.ToString())
          leftPayload = shared
          rightPayload = shared
        } else if kind == 3 {
          leftPayload = String.Concat("equal-", index.ToString())
          rightPayload = String.Concat("equal-", index.ToString())
        }
        let field = kind >= 2 ? StyleField.FontFamily : StyleField.Width
        left[index] = StyleEntry{
          Field: field,
          A: numeric,
          B: numeric + 1.0F,
          C: numeric + 2.0F,
          D: numeric + 3.0F,
          Payload: leftPayload,
        }
        right[index] = StyleEntry{
          Field: field,
          A: numeric,
          B: numeric + 1.0F,
          C: numeric + 2.0F,
          D: rightD,
          Payload: rightPayload,
        }
        index++
      }
      let expected = kind != 1
      let expectedHits = expected ? batch : 0
      index = 0
      while index < warmup {
        let hits = compareBatch(left, right, batch)
        if hits != expectedHits { throw InvalidOperationException("style warmup result mismatch") }
        sink = sink + int64(hits)
        index++
      }
      let times = [samples]int64
      let allocations = [samples]int64
      index = 0
      while index < samples {
        let allocationStart = GC.GetAllocatedBytesForCurrentThread()
        let start = Stopwatch.GetTimestamp()
        let hits = compareBatch(left, right, batch)
        let end = Stopwatch.GetTimestamp()
        let allocationEnd = GC.GetAllocatedBytesForCurrentThread()
        if hits != expectedHits { throw InvalidOperationException("style measured result mismatch") }
        times[index] = ticksToNanoseconds(end - start)
        allocations[index] = allocationEnd - allocationStart
        sink = sink + int64(hits)
        index++
      }
      writeResult("style", name, 0, warmup, samples, batch, times, allocations)
    }

    private func compareBatch(left []StyleEntry, right []StyleEntry, batch int32) int32 {
      var hits int32
      var index int32
      while index < batch {
        let slot = index & 63
        if sameStyleEntry(left[slot], right[slot]) { hits++ }
        index++
      }
      return hits
    }

    private func verifyStyleSemantics() {
      let nan = StyleEntry{ Field: StyleField.Width, A: Single.NaN }
      if sameStyleEntry(nan, nan) {
        throw InvalidOperationException("NaN style comparison must remain unequal")
      }
      let shared = String.Concat("shared", "-semantic")
      let sharedLeft = StyleEntry{ Field: StyleField.FontFamily, Payload: shared }
      let sharedRight = StyleEntry{ Field: StyleField.FontFamily, Payload: shared }
      if !sameStyleEntry(sharedLeft, sharedRight) {
        throw InvalidOperationException("shared text style comparison mismatch")
      }
      let distinctSeed = Stopwatch.GetTimestamp().ToString()
      let distinctLeft = StyleEntry{
        Field: StyleField.FontFamily,
        Payload: String.Concat("distinct-", distinctSeed),
      }
      let distinctRight = StyleEntry{
        Field: StyleField.FontFamily,
        Payload: String.Concat("distinct-", distinctSeed),
      }
      if Object.ReferenceEquals(distinctLeft.Payload, distinctRight.Payload)
        || !sameStyleEntry(distinctLeft, distinctRight) {
          throw InvalidOperationException("distinct equal text style comparison mismatch")
        }
    }

    private func runKeyed(name string, count int32, warmup int32, samples int32) {
      let forward = keyedTree(count, 0, count)
      let alternate = if name == "reorder" {
        keyedTree(count, count / 2, count)
      } else if name == "remove-add" {
        keyedTree(count - 1, 0, count)
      } else {
        forward
      }
      let rec = Reconciler{ Res: Resolver{} }
      let root = rec.Mount(forward)
      let retained = Dictionary[string, Node](count)
      var index int32
      while index < root.Children.Count {
        let child = root.Children[index]
        guard let key = child.Key else { throw InvalidOperationException("mounted child key missing") }
        retained.Add(key, child)
        index++
      }
      verifyKeyed(root, forward, retained, name, true)
      index = 0
      while index < warmup {
        let target = keyedTarget(name, forward, alternate, index)
        rec.Effects = ReconcileEffects.None
        rec.Diff(root, target)
        verifyKeyed(root, target, retained, name, false)
        verifyEffects(rec.Effects, name)
        index++
      }
      let times = [samples]int64
      let allocations = [samples]int64
      index = 0
      while index < samples {
        let target = keyedTarget(name, forward, alternate, warmup + index)
        rec.Effects = ReconcileEffects.None
        let allocationStart = GC.GetAllocatedBytesForCurrentThread()
        let start = Stopwatch.GetTimestamp()
        rec.Diff(root, target)
        let end = Stopwatch.GetTimestamp()
        let allocationEnd = GC.GetAllocatedBytesForCurrentThread()
        times[index] = ticksToNanoseconds(end - start)
        allocations[index] = allocationEnd - allocationStart
        verifyKeyed(root, target, retained, name, false)
        verifyEffects(rec.Effects, name)
        sink = sink + int64(root.Children.Count)
        index++
      }
      TextLayouts.DisposeTree(root)
      writeResult("keyed", name, count, warmup, samples, 1, times, allocations)
    }

    private func keyedTarget(name string, forward Container, alternate Container, ordinal int32) Container {
      if name == "stable" { return forward }
      return (ordinal & 1) == 0 ? alternate : forward
    }

    private func keyedTree(length int32, offset int32, modulus int32) Container {
      let children = List[Blob](length)
      var index int32
      while index < length {
        let value = (index + offset) % modulus
        let key = value.ToString()
        children.Add(Text{ Key: key, Content: key })
        index++
      }
      return Container{ Children: children }
    }

    private func verifyKeyed(root Node, expected Container, retained Dictionary[string, Node], name string, initial bool) {
      if root.Children.Count != expected.Children.Count {
        throw InvalidOperationException("keyed child count mismatch")
      }
      var index int32
      while index < expected.Children.Count {
        guard let key = expected.Children[index].Key else {
          throw InvalidOperationException("expected child key missing")
        }
        let actual = root.Children[index]
        if actual.Key != key || actual.Content != key {
          throw InvalidOperationException("keyed child structure mismatch")
        }
        if name != "remove-add" || index < expected.Children.Count - 1 || expected.Children.Count < retained.Count {
          if retained.TryGetValue(key, out var original) && !Object.ReferenceEquals(actual, original) {
            throw InvalidOperationException("retained keyed child identity changed")
          }
        }
        index++
      }
      if initial && root.Children.Count != retained.Count {
        throw InvalidOperationException("initial keyed identity count mismatch")
      }
    }

    private func verifyEffects(effects ReconcileEffects, name string) {
      let structural = (int32(effects) & int32(ReconcileEffects.Structure)) != 0
      if name == "stable" && structural {
        throw InvalidOperationException("stable keyed diff changed structure")
      }
      if name != "stable" && !structural {
        throw InvalidOperationException("structural keyed diff missed structure effect")
      }
    }

    private func writeResult(category string, name string, count int32, warmup int32,
      samples int32, batch int32, times []int64, allocations []int64) {
        var allocationTotal int64
        for value in allocations { allocationTotal = allocationTotal + value }
        Console.WriteLine("reconcile-benchmark: category=" + category
          + " case=" + name
          + " count=" + count.ToString()
          + " warmup=" + warmup.ToString()
          + " samples=" + samples.ToString()
          + " batch=" + batch.ToString()
          + " cpu_p50_ns=" + percentile(times, 0.50).ToString()
          + " cpu_p95_ns=" + percentile(times, 0.95).ToString()
          + " cpu_p99_ns=" + percentile(times, 0.99).ToString()
          + " cpu_max_ns=" + maximum(times).ToString()
          + " alloc_p50_B=" + percentile(allocations, 0.50).ToString()
          + " alloc_p95_B=" + percentile(allocations, 0.95).ToString()
          + " alloc_p99_B=" + percentile(allocations, 0.99).ToString()
          + " alloc_total_B=" + allocationTotal.ToString())
      }

    private func environmentCount(name string, fallback int32, maximum int32) int32 {
      let text = Environment.GetEnvironmentVariable(name)
      if text == nil || text == "" { return fallback }
      var value int32
      try {
        value = Int32.Parse(text!!)
      } catch (error Exception) {
        throw InvalidOperationException(name + " must be an integer")
      }
      if value < 0 || value > maximum {
        throw InvalidOperationException(name + " is outside the supported range")
      }
      return value
    }

    private func ticksToNanoseconds(ticks int64) int64 ->
      int64(float64(ticks) * 1000000000.0 / float64(Stopwatch.Frequency))

    private func percentile(values []int64, value float64) int64 {
      let sorted = [values.Length]int64
      Array.Copy(values, sorted, values.Length)
      Array.Sort(sorted)
      let raw = int32(Math.Ceiling(float64(sorted.Length) * value)) - 1
      let index = if raw < 0 { 0 } else if raw >= sorted.Length { sorted.Length - 1 } else { raw }
      return sorted[index]
    }

    private func maximum(values []int64) int64 {
      var result int64
      for value in values {
        if value > result { result = value }
      }
      return result
    }
  }
}
